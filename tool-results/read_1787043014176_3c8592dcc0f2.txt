  1095→  1095→-- Generate unique event slug
  1096→  1096→CREATE OR REPLACE FUNCTION generate_event_slug(base_slug TEXT)
  1097→  1097→RETURNS TEXT AS $$
  1098→  1098→DECLARE
  1099→  1099→  final_slug TEXT := base_slug;
  1100→  1100→  counter INT := 0;
  1101→  1101→BEGIN
  1102→  1102→  -- Normalize slug
  1103→  1103→  final_slug := lower(regexp_replace(final_slug, '[^a-z0-9]+', '-', 'g'));
  1104→  1104→  final_slug := regexp_replace(final_slug, '^-|-$', '', 'g');
  1105→  1105→
  1106→  1106→  -- Check uniqueness and append counter if needed
  1107→  1107→  WHILE EXISTS (SELECT 1 FROM events WHERE slug = final_slug) LOOP
  1108→  1108→    counter := counter + 1;
  1109→  1109→    final_slug := base_slug || '-' || counter;
  1110→  1110→  END LOOP;
  1111→  1111→
  1112→  1112→  RETURN final_slug;
  1113→  1113→END;
  1114→  1114→$$ LANGUAGE plpgsql;

  1116→  1116→-- Find similar faces using cosine similarity
  1117→  1117→CREATE OR REPLACE FUNCTION find_similar_faces(
  1118→  1118→  query_embedding vector(128),
  1119→  1119→  event_id_param UUID,
  1120→  1120→  threshold DECIMAL DEFAULT 0.6
  1121→  1121→)
  1122→  1122→RETURNS TABLE (
  1123→  1123→  detection_id UUID,
  1124→  1124→  photo_id UUID,
  1125→  1125→  cluster_id UUID,
  1126→  1126→  confidence DECIMAL,
  1127→  1127→  thumbnail_url TEXT
  1128→  1128→) AS $$
  1129→  1129→BEGIN
  1130→  1130→  RETURN QUERY
  1131→  1131→  SELECT
  1132→  1132→    fd.id AS detection_id,
  1133→  1133→    fd.photo_id,
  1134→  1134→    fd.cluster_id,
  1135→  1135→    (1 - (fd.embedding <=> query_embedding))::DECIMAL(3,2) AS confidence,
  1136→  1136→    p.thumbnail_urls->>'sm' AS thumbnail_url
  1137→  1137→  FROM face_detections fd
  1138→  1138→  JOIN photos p ON fd.photo_id = p.id
  1139→  1139→  WHERE p.event_id = event_id_param
  1140→  1140→    AND p.is_flagged = FALSE
  1141→  1141→    AND (1 - (fd.embedding <=> query_embedding)) >= threshold
  1142→  1142→  ORDER BY fd.embedding <=> query_embedding
  1143→  1143→  LIMIT 100;
  1144→  1144→END;
  1145→  1145→$$ LANGUAGE plpgsql;
