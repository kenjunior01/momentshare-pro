   352→  sample_count         INT DEFAULT 0,
   353→  confidence_score     DECIMAL(3,2),         -- cluster consistency 0-1
   354→
   355→  -- Guest self-identification
   356→  guest_claimed_by     UUID REFERENCES users(id) ON DELETE SET NULL,
   357→  claimed_at           TIMESTAMPTZ,
   358→
   359→  -- Stats
   360→  photo_count          INT DEFAULT 0,
   361→
   362→  metadata             JSONB DEFAULT '{}',
   363→  created_at           TIMESTAMPTZ NOT NULL DEFAULT NOW(),
   364→  updated_at           TIMESTAMPTZ NOT NULL DEFAULT NOW()
   365→);

   370→-- Individual face detections in photos
   371→CREATE TABLE face_detections (
   372→  id              UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
   373→  photo_id        UUID NOT NULL REFERENCES photos(id) ON DELETE CASCADE,
   374→  cluster_id      UUID REFERENCES face_clusters(id) ON DELETE SET NULL,
   375→
   376→  -- ML Detection data
   377→  embedding       vector(128) NOT NULL,
   378→  bounding_box    JSONB NOT NULL,            -- {x, y, width, height} normalized 0-1
   379→  confidence      DECIMAL(3,2) NOT NULL,     -- detection confidence 0-1
   380→  landmark_points JSONB,                    -- {left_eye, right_eye, nose, mouth_left, mouth_right}
   381→  face_quality    DECIMAL(3,2),              -- blur/exposure check
   382→
   383→  -- Clustering status
   384→  is_verified     BOOLEAN DEFAULT FALSE,    -- confirmed by a human
   385→  reviewed_by     UUID REFERENCES users(id),
   386→
   387→  created_at      TIMESTAMPTZ NOT NULL DEFAULT NOW()
   388→);
   389→
   390→CREATE INDEX idx_detections_photo ON face_detections (photo_id);
   391→CREATE INDEX idx_detections_cluster ON face_detections (cluster_id) WHERE cluster_id IS NOT NULL;
   392→CREATE INDEX idx_detections_unassigned ON face_detections (photo_id)
   393→  WHERE cluster_id IS NULL;

   957→-- Maintain photo_count on events
   958→CREATE OR REPLACE FUNCTION update_event_photo_count()
   959→RETURNS TRIGGER AS $$
   960→BEGIN
   961→  IF TG_OP = 'INSERT' THEN
   962→    UPDATE events SET photo_count = photo_count + 1 WHERE id = NEW.event_id;
   963→  ELSIF TG_OP = 'DELETE' THEN
   964→    UPDATE events SET photo_count = photo_count - 1 WHERE id = OLD.event_id;
   965→  END IF;
   966→  RETURN COALESCE(NEW, OLD);
   967→END;
   968→$$ LANGUAGE plpgsql;
   969→
   970→CREATE TRIGGER tr_photo_count AFTER INSERT OR DELETE ON photos
   971→  FOR EACH ROW EXECUTE FUNCTION update_event_photo_count();

   990→-- VIEWS (Analytics & Reporting)
   991→-- ============================================================
   992→
   993→-- Event summary view (for dashboards)
   994→CREATE OR REPLACE VIEW v_event_summary AS
   995→SELECT
   996→  e.id AS event_id,
   997→  e.name,
   998→  e.slug,
   999→  e.event_type,
  1000→  e.status,
  1001→  e.privacy_mode,
  1002→  e.starts_at,
  1003→  e.ends_at,
  1004→  e.photo_count,
  1005→  e.guest_count,
  1006→  e.view_count,
  1007→  e.download_count,
  1008→  e.share_count,
  1009→  e.created_at,
  1010→  u.full_name AS creator_name,
  1011→  u.email AS creator_email,
  1012→  o.name AS org_name,
  1013→  o.plan AS org_plan,
  1014→  -- Aggregate analytics
  1015→  COALESCE(SUM(a.page_views), 0) AS total_page_views,
  1016→  COALESCE(SUM(a.unique_visitors), 0) AS total_unique_visitors,
  1017→  COALESCE(SUM(a.photo_uploads), 0) AS total_uploads,
  1018→  COALESCE(SUM(a.photo_downloads), 0) AS total_downloads,
  1019→  COALESCE(SUM(a.face_scans), 0) AS total_face_scans,
  1020→  COALESCE(SUM(a.qr_scans), 0) AS total_qr_scans,
  1021→  COALESCE(SUM(a.revenue), 0) AS total_revenue,
  1022→  -- Face recognition stats
  1023→  COALESCE(fc.cluster_count, 0) AS unique_faces_detected,
  1024→  COALESCE(gb.entry_count, 0) AS guestbook_entries,
  1025→  -- Collaborator count
  1026→  COALESCE(ec.collab_count, 0) AS collaborator_count
  1027→FROM events e
  1028→LEFT JOIN users u ON e.creator_id = u.id
  1029→LEFT JOIN organizations o ON e.org_id = o.id
  1030→LEFT JOIN event_analytics_daily a ON a.event_id = e.id
  1031→LEFT JOIN (SELECT event_id, COUNT(*) AS cluster_count FROM face_clusters GROUP BY event_id) fc ON fc.event_id = e.id
  1032→LEFT JOIN (SELECT event_id, COUNT(*) AS entry_count FROM guestbook_entries WHERE is_hidden = FALSE GROUP BY event_id) gb ON gb.event_id = e.id
  1033→LEFT JOIN (SELECT event_id, COUNT(*) AS collab_count FROM event_collaborators GROUP BY event_id) ec ON ec.event_id = e.id

  1095→-- Generate unique event slug
  1096→CREATE OR REPLACE FUNCTION generate_event_slug(base_slug TEXT)
  1097→RETURNS TEXT AS $$
  1098→DECLARE
  1099→  final_slug TEXT := base_slug;
  1100→  counter INT := 0;
  1101→BEGIN
  1102→  -- Normalize slug
  1103→  final_slug := lower(regexp_replace(final_slug, '[^a-z0-9]+', '-', 'g'));
  1104→  final_slug := regexp_replace(final_slug, '^-|-$', '', 'g');
  1105→
  1106→  -- Check uniqueness and append counter if needed
  1107→  WHILE EXISTS (SELECT 1 FROM events WHERE slug = final_slug) LOOP
  1108→    counter := counter + 1;
  1109→    final_slug := base_slug || '-' || counter;
  1110→  END LOOP;
  1111→
  1112→  RETURN final_slug;
  1113→END;
  1114→$$ LANGUAGE plpgsql;

  1116→-- Find similar faces using cosine similarity
  1117→CREATE OR REPLACE FUNCTION find_similar_faces(
  1118→  query_embedding vector(128),
  1119→  event_id_param UUID,
  1120→  threshold DECIMAL DEFAULT 0.6
  1121→)
  1122→RETURNS TABLE (
  1123→  detection_id UUID,
  1124→  photo_id UUID,
  1125→  cluster_id UUID,
  1126→  confidence DECIMAL,
  1127→  thumbnail_url TEXT
  1128→) AS $$
  1129→BEGIN
  1130→  RETURN QUERY
  1131→  SELECT
  1132→    fd.id AS detection_id,
  1133→    fd.photo_id,
  1134→    fd.cluster_id,
  1135→    (1 - (fd.embedding <=> query_embedding))::DECIMAL(3,2) AS confidence,
  1136→    p.thumbnail_urls->>'sm' AS thumbnail_url
  1137→  FROM face_detections fd
  1138→  JOIN photos p ON fd.photo_id = p.id
  1139→  WHERE p.event_id = event_id_param
  1140→    AND p.is_flagged = FALSE
  1141→    AND (1 - (fd.embedding <=> query_embedding)) >= threshold
  1142→  ORDER BY fd.embedding <=> query_embedding
  1143→  LIMIT 100;
  1144→END;
  1145→$$ LANGUAGE plpgsql;
